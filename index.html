import 'package:bottle1/uihelper/util.dart';
import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget{
  @override
  Widget build(BuildContext context){
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(
        primarySwatch: Colors.lightGreen,
        textTheme: TextTheme(
          displayLarge: TextStyle(fontSize: 55, color: Colors.red),
          bodyMedium: TextStyle(fontSize: 15, color: Colors.green)
        )
      ),
      home: Scaffold(
        appBar: AppBar(
          title: Text("Dashboard"),
        ),
        body: Column(
          children: [
            Text("Hellow 1", style: Theme.of(context).textTheme.displayLarge),
            Text("Hellow 2", style: Theme.of(context).textTheme.displayLarge!.copyWith(color: Colors.yellow)),
            Text("Hellow 3", style: TextStyle(fontSize: 20, color: Colors.green),),
            Text("Hellow 4", style: TextStyle(fontSize: 20, color: Colors.blue),),
            Text("Hellow 5", style: Theme.of(context).textTheme.bodyMedium),
            Text("Hellow 6", style: aaa()),
            Text("Hellow 7", style: bbb(textcolor1: Colors.pink)),
          ],
        )
      ),
    );
  }
}
