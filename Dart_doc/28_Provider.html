Provider

Provider


Provider = a box that stores data
Any screen can read or change that data

PROVIDER HAS ONLY 3 IMPORTANT PARTS
-----------
- Model (Data + Logic)
- Provider (Wrap your app)
- Consumer (Use the data)

Step1: Add provider(pubspec.yaml)
----------------
dependencies:
  provider: ^6.0.5
  
  
STEP 2: Create a Provider class(counter_provider.dart)
------------------------------
import 'package:flutter/material.dart';

class CounterProvider extends ChangeNotifier {
  int count = 0;

  void increment() {
    count++;
    notifyListeners(); 
  }
}

//ChangeNotifier - something changed, rebuild yourself
//notifyListeners(); -- 🔔 Hey UI, data changed, rebuild!, refresh UI


STEP 3: Wrap your app with Provider
----------------------------------
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';
import 'counter_provider.dart';
import 'home.dart';

void main() {
  runApp(
    ChangeNotifierProvider(
      create: (_) => CounterProvider(),
      child: MyApp(),
    ),
  );
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: HomeScreen(),
    );
  }
}



CounterProvider is available everywhere in the app
//ChangeNotifierProvider() - calls the provider to your app
//create: (_) => CounterProvider()

STEP 4: Use Provider in UI
------------------------------
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';
import 'counter_provider.dart';

class HomeScreen extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    final counter = Provider.of<CounterProvider>(context);

    return Scaffold(
      appBar: AppBar(title: Text("Provider Easy")),
      body: Center(
        child: Text(
          counter.count.toString(),
          style: TextStyle(fontSize: 40),
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: () {
          counter.increment();
        },
        child: Icon(Icons.add),
      ),
    );
  }
}





//Provider.of<CounterProvider>(context) --- Get shared data
//counter - variable got the provider data
//counter.count	Read data, The value is 0
//counter.increment()	--- Change data
//notifyListeners()	--- UI rebuilds
//WITHOUT Consumer Even if you call notifyListeners(); UI will NOT update
//ChangeNotifier sends signal
   Consumer receives signal
   setState is NOT needed

3.home_screen.dart (Consumer usage)
--------------------------------
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';
import 'counter_provider.dart';

class HomeScreen extends StatelessWidget {
  const HomeScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text("Provider + Consumer"),
      ),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [

            /// 🔴 Consumer LISTENS here
              Consumer<CounterProvider>(
              builder: (context, counter, child) {  
                return Text(
                  counter.count.toString(),
                  style: const TextStyle(
                    fontSize: 40,
                    fontWeight: FontWeight.bold,
                  ),
                );
              },
            ),

            const SizedBox(height: 20),

            ElevatedButton(
              onPressed: () {
                  context.read<CounterProvider>().increment();  
              },
              child: const Text("Increment"),
            ),
          ],
        ),
      ),
    );
  }
}
