5. insert in particular index using - insert, insertAll to the list-


void main () {
  var listNames = [10, 20, 30, 40];

   name1.insert(1,300);

10. Spread operator -

void main() {
  List<int> numbers = [1, 2, 3];       // a list of numbers
  print(numbers[0]); // prints 1 (first item)

  List<int>  more = [0, ...numbers, 4]; // spread operator ... copies values from numbers
  print(more); // prints [0, 1, 2, 3, 4]

  List<int> ? maybeNumbers;
  List<int>  safe = [9, ...(maybeNumbers ?? [])]; // avoid null with ?? []
  print(safe);
}


o/p
====
1
[0, 1, 2, 3, 4]
[9]
