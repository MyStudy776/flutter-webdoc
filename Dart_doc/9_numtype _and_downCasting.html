num type and down casting

1. num type and down casting - (as)

void main() {
  num x = 5;
  int y = x; //error, a num type variable can not assign to an int type variable.
  print(y);
}



===================

void main() {
  num x = 5;
  int y = x as int; //this is down casting
  print(y);
}
