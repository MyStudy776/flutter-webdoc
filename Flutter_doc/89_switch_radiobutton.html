1.Example of Switch 

 
import 'package:flutter/material.dart';

class Checkboxfile extends StatefulWidget {
  const Checkboxfile({super.key});

  @override
  State<Checkboxfile> createState() => _CheckboxfileState();
}

class _CheckboxfileState extends State<Checkboxfile> {

  bool isconnected = false;

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Switch(
        value: isconnected,
        activeThumbColor: Colors.blue,
        inactiveThumbColor: Colors.pinkAccent,
        onChanged: (value){
          setState(() {
            isconnected = value;
          });
        }
        ),
  
    );
  }
}

4.Example of auto generated list tile 

 
import 'package:flutter/material.dart';

class Checkboxfile extends StatefulWidget {
  const Checkboxfile({super.key});

  @override
  State<Checkboxfile> createState() => _CheckboxfileState();
}

class _CheckboxfileState extends State<Checkboxfile> {

  // A list for switch states
  List<bool> switchValues = List.generate(10, (index) => false);
  // You can change 10 → any number

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: ListView.builder(
        itemCount: switchValues.length,
        itemBuilder: (context, index) {
          return SwitchListTile(
            value: switchValues[index],
            onChanged: (value) {
              setState(() {
                switchValues[index] = value; // change only this switch
              });
            },
            title: Text("Program ${index + 1}"),
            subtitle: Text("Testing item number ${index + 1}"),
          );
        },
      ),
    );
  }
}

5.Example of Listview builder 

 
import 'package:flutter/material.dart';

class Checkboxfile extends StatefulWidget {
  const Checkboxfile({super.key});

  @override
  State<Checkboxfile> createState() => _CheckboxfileState();
}

class _CheckboxfileState extends State {

  // Your title + subtitle list
  List<Map<String, String>> items = [
    {"title": "First Program", "subtitle": "Testing the code"},
    {"title": "Second Program", "subtitle": "Testing the fish"},
    {"title": "Third Program", "subtitle": "Working on UI"},
    {"title": "Fourth Program", "subtitle": "Switch Testing"},
    {"title": "Fifth Program", "subtitle": "Final Output"},
  ];

  // List<bool> will handle toggle states
  late List<bool> switchValues;

  @override
  void initState() {
    super.initState();
    switchValues = List.filled(items.length, false); // auto-size based on items
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: ListView.builder(
        itemCount: items.length,
        itemBuilder: (context, index) {
          return SwitchListTile(
            value: switchValues[index],
            onChanged: (value) {
              setState(() {
                switchValues[index] = value;
              });
            },
            title: Text(items[index]["title"]!),
            subtitle: Text(items[index]["subtitle"]!),
          );
        },
      ),
    );
  }
}
