ListView widget -

1. List view -

It will automaticall scroll horizontally and vertically when content is beyond the screen.

Difference -

SingleChildScrollView expects a single child, which is typically a column, row, or another widget.
ListView can have multiple children, and it's designed to handle long lists efficiently.
Efficiency for Large Data Sets:

SingleChildScrollView renders all its children at once, which can be inefficient for a large number of items.
ListView only renders the items that are currently visible on the screen, making it much more efficient for long lists.
Usage Scenario:

Use SingleChildScrollView when you have a single child that needs to be scrollable (e.g., a form with various input fields).
Use ListView when you have a list of items (e.g., messages in a chat app, articles in a news app) and you want to efficiently scroll through them.

4. Listview Builder -

- It will manage when large amount of data will be inserted.
- It will build the list view.
Here we will define how one item how it looks like. According to that all the items in array will be changed.
we will get an index in ItemBuilder.- ListView.builder(itemBuilder: itemBuilder)
or we can use context - ListView.builder(itemBuilder: (context, index)
itemcount---> Specifies howmany item need to build.

5. Listview Builder Example 1 -

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
          appBar: AppBar(
            title: Text('Hello'),
            backgroundColor: Colors.orange,
          ),
          body: ListView.builder(itemBuilder: (context, index){
            return Text('one', style: TextStyle(fontSize: 21, fontWeight: FontWeight.w500),);
           },
          itemCount: 5,
      ),
    ),
    );
  }
}

7. Listview Builder Example 3-

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    var arrNames = ['Rajesh','Monika', 'Raman','Kohli', 'Birat', 'Ranveer'];
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(
          title: Text('Hello'),
          backgroundColor: Colors.orange,
        ),
        body: ListView.builder(itemBuilder: (context, index){
          return Text(arrNames[index], style: TextStyle(fontSize: 21, fontWeight: FontWeight.w500),);

          itemCount: arrNames.length,
          itemExtent: 100, ----->it will give space between items
        ),
      ),
    );
  }
}



itemExtent: 100, --->After this space came between items. when items are out of screen, then these will be scrollable.
scrollDirection: Axis.horizontal, --->items are in horizontal direction

Advantage -

when scrolling the listview builder item, it will destroy the the item from memory, which are not visible on the screen. So always the memory is recycled.

8. Listview Separator Example 1


import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    var arrNames = ['Rajesh','Monika', 'Raman','Kohli', 'Birat', 'Ranveer','Rajesh','Monika', 'Raman','Kohli', 'Birat', 'Ranveer'];
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(
          title: Text('Hello'),
          backgroundColor: Colors.orange,
        ),
        body: ListView.separated(itemBuilder: (context, index){
          return Text(arrNames[index], style: TextStyle(fontSize: 21, fontWeight: FontWeight.w500),);
        },
          itemCount: arrNames.length,
          separatorBuilder: (context, index){
          return Divider(height: 100, thickness: 2,);   
          },
        ),
      ),
    );
  }
}





return Divider(height: 100, thickness: 2,); --->The items are separated by a divider. 
                                            Here divider is the separataor.
                                            We can put any widget as a separator.

11. Listview Builder Example

import 'package:flutter/material.dart';
void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});


  @override
  Widget build(BuildContext context) {

    return MaterialApp(
        title: 'Flutter Demo',
        theme: ThemeData(
          //brightness: Brightness.dark,
          //primaryColor: Colors.cyan,
  colorScheme: ColorScheme.fromSeed(seedColor: Colors.deepPurple),
          useMaterial3: true,
        ),
        home: MyHomePage()
    );
  }
}

class MyHomePage extends StatefulWidget {

  @override
  State<MyHomePage> createState() => _MyHomePageState();
}

class _MyHomePageState extends State<MyHomePage> {
  List<String> fruits = ['Orange', 'Apple', 'Mango', 'Banana','Orange', 'Apple', 'Mango', 'Banana'];
  Map fruits_person = {
    'fruits': ['Orange', 'Apple', 'Mango', 'Banana','Orange', 'Apple', 'Mango', 'Banana'],
    'names': ['Karan', 'Akshit', 'Gaurav', 'Aasif','Karan', 'Akshit', 'Gaurav', 'Aasif'],
  };
  @override
  Widget build(BuildContext context) {

    return Scaffold(

      appBar: AppBar(
        title: Text('List view'),
        backgroundColor: Colors.cyan,
        elevation: 0,
      ),
      body: Container(
        child: ListView.builder(
          itemCount: fruits.length,
             itemBuilder: (context, index) {
               return Card(
                 child: ListTile(
                   onTap: () {
                     print((fruits_person['fruits'][index]));
                   },
                   leading: Icon(Icons.person),
                   title: Text(fruits_person['fruits'][index]),
                   subtitle: Text(fruits_person['names'][index]),
                 ),
               );
             },
           ),


      )
    );
  }
}
