Container Widget

1. Container -

It is a widget that combines common positioning and sizing of the child widgets.
It is a box of storing contents.

2. Create a stateless widget -

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp1());
}

class MyApp1 extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(),
    );
  }
}

3. Create a Container, add a text, make the container as center -

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp1());
}

class MyApp1 extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(
          title: Text(
            "My Home page",
            style:
                TextStyle(color: Colors.pink[900], fontWeight: FontWeight.bold),
          ),
        ),
        body: Center(
          child: Container(
            width: 100,
            height: 100,
            color: Colors.amber,
            child: Text('Hello Dobi!'),
          ),
        ),
      ),
    );
  }
}

4. keep center the text inside the container -

body: Center(
          child: Container(
            width: 100,
            height: 100,
            color: Colors.amber,
            child: Center(child: Text('Hello Dobi!', style: TextStyle(color: Colors.white),)),
          ),
        ),
    );
  }
}

5. Styling the text -

body: Center(
          child: Container(
            width: 100,
            height: 100,
            color: Colors.amber,
            child: Center(child: Text('Hello Dobi!', style: TextStyle(
                color: Colors.white,
              fontSize: 25,
              fontWeight: FontWeight.bold,
              backgroundColor: Colors.black
            ),)),
          ),
        ),
    );
  }
}

6. Add margin to the container -

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget{
  @override
  Widget build(BuildContext context){
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(
          title: Text("Dashboard"),
        ),
        body: Container(
          color: Colors.blue,
            margin: EdgeInsets.all(11),
            child: Padding(
              padding: const EdgeInsets.all(15),
              child: Text("Hello World", style: TextStyle(fontSize: 25, color: Colors.white),),
            )),
      ),
    );
  }
}
