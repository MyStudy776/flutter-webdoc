3. home.dart - Use media query

import 'package:flutter/material.dart';

class Home extends StatefulWidget {
  const Home({super.key});

  @override
  State<Home> createState() => _HomeState();
}

class _HomeState extends State<Home> {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Stack(
        children: [
          Container(
            width: MediaQuery.of(context).size.width,----------->It will use the full width of the screen
            height: MediaQuery.of(context).size.height,-------->It will use the full height of the screen
            decoration: BoxDecoration(
              gradient: LinearGradient(
                colors: [
                  Color(0xff213A50),
                  Color(0xff071938)
                ]
              )
            ),
          ),
          SafeArea(child: Text("I am Home", style: TextStyle(color: Colors.white),)),--->safearea used -->Text will be written under the navigator page
        ],
      ),
    );
  }
}

4. home.dart - Use Boxdecoration, Linear gradient, stack, searchbox

import 'package:flutter/material.dart';

class Home extends StatefulWidget {
  const Home({super.key});

  @override
  State<Home> createState() => _HomeState();
}

class _HomeState extends State<Home> {

  TextEditingController searchController = new TextEditingController();
  
  @override
  Widget build(BuildContext context) {
    return Scaffold(
        body: Stack(---->one widget placed on top of another widget, LIFO- last in first out.
      children: [
        Container(------------->This is screen area container.
          width: MediaQuery.of(context).size.width,
          height: MediaQuery.of(context).size.height,
          decoration: BoxDecoration(
            gradient:
                LinearGradient(colors: [Color(0xff213A50), Color(0xff071938)]),
          ),
        ),

        
        Column(
          children: [
            SafeArea(
              child: Container(----------->Search Bar container on top of 1st container
                padding: EdgeInsets.symmetric(horizontal: 8),
                margin: EdgeInsets.symmetric(horizontal: 24, vertical: 20),
                decoration: BoxDecoration(
                  color: Colors.white,
                  borderRadius: BorderRadius.circular(24),
                ),
                child: Row(
                  children: [
                    GestureDetector(
                      onTap: () {
                        if ((searchController.text).replaceAll(" ", "") == "") {
                          print("Blank search");
                        } else {
                          Navigator.pushReplacementNamed(context, "/loading",
                              arguments: {
                                "searchText": searchController.text,
                              });
                        }
                      },
                      child: Container(--------->container for icon search
                        child: Icon(
                          Icons.search,
                          color: Colors.blueAccent,
                        ),
                        margin: EdgeInsets.fromLTRB(3, 0, 7, 0),
                      ),
                    ), //GestureDetector
                    Expanded(
                      child: TextField(------------->for Text Field
                        controller: searchController,
                        decoration: InputDecoration(
                            border: InputBorder.none,
                            hintText: "Let's Cook Something!"),
                      ),
                    )
                  ],
                ),//Row
              ),
            ),
             Container( 
              padding: EdgeInsets.symmetric(horizontal: 20),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    "WHAT DO YOU WANT TO COOK TODAY?",
                    style: TextStyle(fontSize: 33, color: Colors.white),
                  ),
                  SizedBox(
                    height: 10,
                  ),
                  Text(
                    "Let's Cook Something New!",
                    style: TextStyle(fontSize: 20, color: Colors.white),
                  )
                ],
              ),
            )//Container
          ],
        ),//Column
      ],
    ),//Stack
	);//Scaffold
  }
}
