import 'package:flutter/cupertino.dart';
import 'package:flutter/material.dart';
import 'package:shikar/screens/drawerWithBottomNav/home_screen.dart';
import 'package:shikar/screens/drawerWithBottomNav/setting_screen.dart';
import 'package:shikar/screens/drawerWithBottomNav/profile_screen.dart';


class DrawerwithFile extends StatefulWidget {
  const DrawerwithFile({super.key});

  @override
  State createState() => _DrawerwithFileState();
}

class _DrawerwithFileState extends State {

  List widgets = [
    Homescreen(),
    ProfileScreen(),
    Steeing_screen()
    ];

    int currentIndex1 = 0;

    GlobalKey globalKey = GlobalKey();


  @override
  Widget build(BuildContext context) {
    return Scaffold(

      key: globalKey,

      appBar: AppBar(
        title: Text("Drawer with"),
      ),


      body: widgets[currentIndex1],

     

      drawer: Drawer(
        child: ListView(
          children: [
            SizedBox(height: 50,),
            FlutterLogo(size: 100,),
            SizedBox(height: 20,),
            ListTile(
              //selected: true,
              selected: currentIndex1==0,
              selectedColor: Colors.amber,
              onTap: (){
                setState(() {
                  currentIndex1 = 0;

                  globalKey.currentState?.closeDrawer();
                });
              },
              title: Text("Home"),
              leading: Icon(Icons.home),
            ),

            ListTile(
              //selected: true,
              selected: currentIndex1==1,
              selectedColor: Colors.amber,
              onTap: (){
                setState(() {
                  currentIndex1 = 1;
                  globalKey.currentState?.closeDrawer();
                });
              },
              title: Text("Profile"),
              leading: Icon(CupertinoIcons.profile_circled),
            ),


            ListTile(
              selected: currentIndex1==2,
              selectedColor: Colors.amber,
              onTap: (){
                setState(() {
                  
                  currentIndex1 = 2;
                  globalKey.currentState?.closeDrawer();
                });
              },
              title: Text("Settings"),
              leading: Icon(Icons.settings),
            )


          ],
        ),
      ),



      /////////////////
      bottomNavigationBar: BottomNavigationBar(
        currentIndex: currentIndex1,
        onTap: (value) {
          print ('The value is $value');
          setState(() {
            currentIndex1 = value;
          });
        },
        items: [
          BottomNavigationBarItem(icon: Icon(Icons.home), label: 'Home'),
          BottomNavigationBarItem(icon: Icon(CupertinoIcons.profile_circled), label: 'Profile'),
          BottomNavigationBarItem(icon: Icon(Icons.settings), label: 'Settings'),
        ]),
    );
  }
}
