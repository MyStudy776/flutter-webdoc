import 'package:flutter/material.dart';
import 'package:mama1/models/catalog.dart';
import 'package:mama1/widget/items_widget.dart';

class ListviewbuilderPage extends StatelessWidget {
  const ListviewbuilderPage({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text("Catalog App"),
      ),

      body: ListView.builder(
        itemCount: CatalogModel.items.length,
        itemBuilder: (context, index) {
		//iterate each item from CatalogModel and Pass it to Itemwidget
          return Itemwidget(item: CatalogModel.items[index],);


        },
      ),
    );
  }
}

5. lib/items_widget.dart -

import 'package:flutter/material.dart';
import 'package:mama1/models/catalog.dart';

class Itemwidget extends StatelessWidget {

  final Item item;
  const Itemwidget({super.key, required this.item});

  @override
  Widget build(BuildContext context) {
    return Card(
      //elevation: 0,
      child: ListTile(
        leading: CircleAvatar(
          backgroundColor: Colors.deepPurple,
          child: Text(item.id.toString(),style: const TextStyle(color: Colors.white),),
        ),
        title: Text(item.name),
        subtitle: Text(item.desc),
        trailing: Text("\$${item.price}"),
      ),
    );

  }
}

import 'package:flutter/material.dart';
import 'package:mama1/utils/routes.dart';

class StartPage extends StatelessWidget {
  const StartPage({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Start Page'),
      ),
      body: Column(
        mainAxisAlignment: MainAxisAlignment.center,
        children: [
          Center(
            child: ElevatedButton(
              onPressed: () {
               // Navigator.pushNamed(context, '/signup');
                Navigator.pushNamed(context, Myroute.signupPage);
              },
              child: const Text('Go to SignUp '),
            ),
          ),

          ElevatedButton(
            onPressed: () {
             // Navigator.pushNamed(context, '/testpage');
              Navigator.pushNamed(context, Myroute.testPage);
            },
            child: const Text('Go to Test Page '),
          ),

          ElevatedButton(
            onPressed: () {
             
              Navigator.pushNamed(context, Myroute.drawerpage);
            },
            child: const Text('Go to drwaer wala Page '),
          ),

          ElevatedButton(
            onPressed: () {
             
              Navigator.pushNamed(context, Myroute.catalogPage); 
            },
            child: const Text('Go to catalog Page '),
          ),
        ],
      )
    );
  }
}
