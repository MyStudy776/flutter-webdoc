Padding and Margin -

1. Padding -

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      theme: ThemeData.dark(),
      home: Scaffold(
          appBar: AppBar(
            title: Text("Dashboard"),
          ),
          body:

          Padding(
            padding: const EdgeInsets.only(top: 11, left: 13),
            child: Text(
              'Hello World',
              style: TextStyle(fontSize: 20,),

            ),
          ),
      ),
    );
  }
}



===
EdgeInsets.all(20) ---> Equal padding from all side

2. Margin -

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      theme: ThemeData.dark(),
      home: Scaffold(
          appBar: AppBar(
            title: Text("Dashboard"),
          ),
          body:

          Container(
            color: Colors.blueGrey,
            margin: EdgeInsets.all(20),
            child: Text(
              'Hello World',
              style: TextStyle(
                fontSize: 20,
              color: Colors.white,),

            ),
          ),
      ),
    );
  }
}

Note - We can find margin inside container.

3. Example -

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      theme: ThemeData.dark(),
      home: Scaffold(
          appBar: AppBar(
            title: Text("Dashboard"),
          ),
          body:

          Container(
            color: Colors.blueGrey,
            margin: EdgeInsets.all(20),
            child: Padding(
              padding: const EdgeInsets.all(20),
              child: Text(
                'Hello World',
                style: TextStyle(
                  fontSize: 20,
                color: Colors.white,),

              ),
            ),
          ),
      ),
    );
  }
}
