import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});


  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flutter Demo',
      theme: ThemeData(
        colorScheme: ColorScheme.fromSeed(seedColor: Colors.deepPurple),
        useMaterial3: true,
      ),
      home: MyHomePage(),
    );
  }
}

class MyHomePage extends StatefulWidget {
  @override
  State createState() => _MyHomePageState();
}

class _MyHomePageState extends State {

  var _width = 200.0;
 var _height = 100.0;
  var color1 = Colors.blueGrey;
  bool flag = true;


  @override
  Widget build(BuildContext context) {
    return Scaffold(
        appBar: AppBar(
          backgroundColor: Theme.of(context).colorScheme.inversePrimary,
          title: Text('Foo Animation'),
        ),
        body: Center(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              Container( -------------->normal container
                width: _width,
                height: _height,
                color: color1,
              ),
              ElevatedButton(onPressed: (){

                setState(() {
                  if(flag){---------------->1st time the flag is true, once button pressed, width, height, color and flag value changed and display
                    _width = 100.0;
                    _height = 200.0;
                    color1 = Colors.yellow;
                    flag = false;
                  }
                  else {------------------>when again button pressed,flag value is false, so else will execute.
                    _width = 200.0;
                    _height = 100.0;
                    color1 = Colors.blueGrey;
                    flag = true;
                  }
                });
              }, child: Text('Animate'))
            ],
          ),
        )

    );
  }
}

5. Animated Container Example- button shape changes, when pressed

import 'package:flutter/material.dart';

class Testpage extends StatefulWidget {
  const Testpage({super.key});

  @override
  State<Testpage> createState() => _TestpageState();
}


//2ndpart
class _TestpageState extends State<Testpage> {

  bool changedbutton = false ;

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text("Test Page"),
      ),
      body: Column(
        children: [
          TextFormField(
            decoration: InputDecoration(
              hintText: 'Enter something',
              labelText: 'Test Input',
              border: OutlineInputBorder(),
            ),
            
          ),


          //Inkwell is used so that container will behave like a clickable button
          InkWell(

            onTap: () => setState(() {
              changedbutton = true;
            }),


            child: AnimatedContainer(
              duration: Duration(seconds: 1),
              width: changedbutton ? 80 : 150,
              height: changedbutton ? 50 : 40,            
              alignment: Alignment.center,
              decoration: BoxDecoration(
                borderRadius: BorderRadius.circular(changedbutton ? 11 : 8),
                color: Colors.deepPurple,
               ),
              child: Text("Submit",style: TextStyle(color: Colors.white,fontWeight: FontWeight.bold,fontSize: 18),),
            ),
          )

         
        ],
      ),
    );
  }
}
