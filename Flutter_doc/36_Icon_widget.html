Icon Widget-

1. Example Icon Widget -

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {

  @override
  Widget build(BuildContext context) {

    return MaterialApp(
      home: Scaffold(
          appBar: AppBar(title: Text('My home page')),
          body: Center(child: Icon(
            Icons.access_alarms,
            size: 100,
          color: Colors.blue,))
      ),
    );
  }
}

2. Install font_awesome package -

Add in pubspec.yaml file - font_awesome_flutter: ^10.6.0
You will get additional icons using this package.

import 'package:flutter/material.dart';
import 'package:font_awesome_flutter/font_awesome_flutter.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {

  @override
  Widget build(BuildContext context) {

    return MaterialApp(
      home: Scaffold(
          appBar: AppBar(title: Text('My home page')),
          body: Wrap(
            direction: Axis.vertical,
            children: [
               Icon(
                Icons.access_alarms,
                size: 100,
                color: Colors.blue,),
              SizedBox(
                height: 30,
              ),
              FaIcon(FontAwesomeIcons.amazon,
              color: Colors.red,
              size:  100,),
            ],

          )
      ),
    );
  }
}
