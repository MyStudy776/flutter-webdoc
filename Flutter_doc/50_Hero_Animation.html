import 'dart:async';

import 'package:disk1/detail_page1.dart';
import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});


  @override
  Widget build(BuildContext context) {
    return MaterialApp(
        title: 'Flutter Demo',
        theme: ThemeData(

          colorScheme: ColorScheme.fromSeed(seedColor: Colors.deepPurple),
          useMaterial3: true,
        ),
        home: MyHomePage()
    );
  }
}

class MyHomePage extends StatelessWidget {

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(

        title: Text("Made by Dobi And Vint"),
      ),
      body: Container(
          child: Center(
            child: InkWell(
              onTap: (){
                Navigator.push(context, MaterialPageRoute(builder: (context) => DetailPage()));
              },
              child: Hero(
                tag: 'background',
                child: Container(
                  height: 100,
                  width: 150,
                  color: Colors.blueAccent,
                ),
              ),
            ),
          )
      ),
    );
  }
}
