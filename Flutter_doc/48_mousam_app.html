worker.dart


import 'package:http/http.dart' as http;
import 'dart:convert';

class worker {

  String location ;

  //Constructor -------------->create one constructor
  worker({this.location = ''})---->we have given null value, because we get error: The parameter 'location' can't have a value of 'null' because of its type 'String'.
  {
    location = this.location;
  }


  String temp = "";
  String humidity = "";
  String air_speed = "";
  String description = "";
  String main = "";

  //method
  Future getData() async--->Future used---> as it will run in background, while other operations are happening.
  {
    //http.Response resp1 = await http.get(Uri.parse("https://api.openweathermap.org/data/2.5/weather?q=Delhi,india&APPID=79b3580ca659ec3db5187e2b4cf2b280"));
    http.Response resp1 = await http.get(Uri.parse("https://api.openweathermap.org/data/2.5/weather?q=$location,india&APPID=79b3580ca659ec3db5187e2b4cf2b280"));
    Map data = jsonDecode(resp1.body);


    //Getting Temp,Humidity
    Map temp_data = data['main'];
    String getHumidity = temp_data['humididty'].toString();
    double getTemp = temp_data['temp'];

    //air speed
    Map wind = data['wind'];
    double getAir_speed = wind["speed"];

    //Getting Description
    List weather_data = data['weather'];
    Map weather_main_data = weather_data[0];
    String getMain_des = weather_main_data['main'];
    String getDesc = weather_main_data["description"];

    temp = getTemp.toString();
    humidity = getHumidity;
    air_speed = getAir_speed.toString();
    description = getDesc;
    main = getMain_des;
  }
}

import 'package:http/http.dart' as http;
import 'dart:convert';

class worker {

  String location ;

  //Constructor
  worker({this.location = ''})
  {
    location = this.location;
  }


  String temp = "";
  String humidity = "";
  String air_speed = "";
  String description = "";
  String main = "";

  //method
  Future getData() async
  {

    try{
      //http.Response resp1 = await http.get(Uri.parse("https://api.openweathermap.org/data/2.5/weather?q=Delhi,india&APPID=79b3580ca659ec3db5187e2b4cf2b280"));
      http.Response resp1 = await http.get(Uri.parse("https://api.openweathermap.org/data/2.5/weather?q=$location,india&APPID=79b3580ca659ec3db5187e2b4cf2b280"));
      Map data = jsonDecode(resp1.body);


      //Getting Temp,Humidity
      Map temp_data = data['main'];
      String getHumidity = temp_data['humididty'].toString();
      double getTemp = temp_data['temp'];

      //air speed
      Map wind = data['wind'];
      double getAir_speed = wind["speed"];

      //Getting Description
      List weather_data = data['weather'];
      Map weather_main_data = weather_data[0];
      String getMain_des = weather_main_data['main'];
      String getDesc = weather_main_data["description"];

      temp = getTemp.toString();
      humidity = getHumidity;
      air_speed = getAir_speed.toString();
      description = getDesc;
      main = getMain_des;

    }catch(e){
      temp = "Invalid temperature";
      humidity = "Invalid humidity value";
      air_speed = "Invalid Air speed value";
      description = "Invalid description";
      main = "Invalid main";

    }

  }
}

import 'package:http/http.dart' as http;
import 'dart:convert';

class worker {

  String location ;

  //Constructor
  worker({this.location = ''})
  {
    location = this.location;
  }


  String temp = "";
  String humidity = "";
  String air_speed = "";
  String description = "";
  String main = "";
  String icon = "";

  //method
  Future getData() async
  {

    try{
      //http.Response resp1 = await http.get(Uri.parse("https://api.openweathermap.org/data/2.5/weather?q=Delhi,india&APPID=79b3580ca659ec3db5187e2b4cf2b280"));
      http.Response resp1 = await http.get(Uri.parse("https://api.openweathermap.org/data/2.5/weather?q=$location,india&APPID=79b3580ca659ec3db5187e2b4cf2b280"));
      Map data = jsonDecode(resp1.body);

      print(data);
      //Getting Temp,Humidity
      Map temp_data = data['main'];
      String getHumidity = temp_data['humididty'].toString();
      double getTemp = temp_data['temp'] - 273.15;

      //air speed
      Map wind = data['wind'];
      double getAir_speed = wind["speed"];

      //Getting Description
      List weather_data = data['weather'];
      Map weather_main_data = weather_data[0];
      String getMain_des = weather_main_data['main'];
      String getDesc = weather_main_data["description"];
      //String icon = weather_main_data["icon"].toString();

      temp = getTemp.toString();
      humidity = getHumidity;
      air_speed = getAir_speed.toString();
      description = getDesc;
      main = getMain_des;
      icon = weather_main_data["icon"].toString();

    }catch(e){
      temp = "Invalid temperature";
      humidity = "Invalid humidity value";
      air_speed = "Invalid Air speed value";
      description = "Invalid description";
      main = "Invalid main";
      icon = "03n";

    }

  }
}
