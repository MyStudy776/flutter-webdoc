2. my_profile_screen.dart - Inside lib folder create one dart file -

import 'package:flutter/material.dart';

class ProfileScreen extends StatelessWidget {
  const ProfileScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
      title: Text("My profile"),
    ),
    body: Container(
      color: Colors.blueGrey.shade200,
      child: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Text("welcome", style: TextStyle(fontSize: 34, color: Colors.green),),
            ElevatedButton(
              onPressed: () {
                Navigator.pop(context);--->when button is pressed, it will navigate to the previous screen, from where it generated.
              },
              child: Text("back"),
            ),
          ],
        ),
      ),
    ),);
  }
}



Navigator.pop is used to remove the current route or screen from the navigation stack and return to the previous screen. This method is typically used when you want to navigate back to the previous screen in your app.

Passing Data from one screen to another screen

3. my_profile_screen.dart 

import 'package:flutter/material.dart';

class ProfileScreen extends StatelessWidget {
  var nameFromHome ;

  ProfileScreen(this.nameFromHome);--------->create one constructor, constructor always with the same name as class.
  //const ProfileScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
      title: Text("My profile"),
    ),
    body: Container(
      color: Colors.blueGrey.shade200,
      child: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Text("welcome, $nameFromHome", style: TextStyle(fontSize: 34, color: Colors.green),),

            ElevatedButton(
              onPressed: () {
                Navigator.pop(context);
              },
              child: Text("back"),
            ),
          ],
        ),
      ),
    ),);
  }
}


nameFromHome ---->argument will be passed when called the class

4. main.dart 

import 'package:disk1/splash_screen.dart';
import 'package:flutter/material.dart';

import 'my_profile_screen.dart';


void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flutter Demo',
      theme: ThemeData(
        primarySwatch: Colors.orange,
      ),
      home: MyHomePage(),
    );
  }
}

class MyHomePage extends StatelessWidget {
  var nameController = TextEditingController();
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('Classico'),
      ),
      body: Center(
        child: Container(
          width: 300,
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              Text('Hello World',
              style: TextStyle(fontSize: 25),),
              SizedBox(height: 11,),
              TextField(
                controller: nameController,
              ),
              SizedBox(height: 11,),

              ElevatedButton(onPressed: () {
                Navigator.push(
                    context,
                  MaterialPageRoute(
                      builder: (context) => ProfileScreen(nameController.text.toString()),),
                );
              }, child: Text('My profile'))
            ],
          ),
        ),
      ),
      backgroundColor: Colors.yellow,
    );
  }

}



nameController---> when button will be pressed, whatever input we wil give in the text fileld, it will passed an argument for class ProfileScreen.
And in the profilescreen page,  the value is stored in variable nameFromHome by the help of constructor the text will be displayed - welcome, $nameFromHome
