Card Widget-

1. Example - without a card widget -

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(
          title: Text('Hello'),
        ),
        body: Text('Hello World'),
      ),
    );
  }
}

2. Example - using Card widget - wrap text with card

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(
          title: Text('Hello'),
        ),
        body: Card(
            child: Text('Hello World', style: TextStyle(fontSize: 22),)),
      ),
    );
  }
}





Note -
Here we wrapped card widget with text widget. We can wrap card with any widget.

3. Example - Give some padding to the text - wrap text with padding 

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(
          title: Text('Hello'),
        ),
        body: Card(
            
            child: Padding(
              padding: const EdgeInsets.all(8.0),
              child: Text('Hello World', style: TextStyle(fontSize: 22),),
            )),
      ),
    );
  }
}

4. Example - Give elevation in the card widget -

            )),
      ),
    );
  }
}

6. Example - card widget -

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(
          title: Text('Hello'),
        ),
        body: Container(
          width: double.infinity,
          child: Card(
            color: Colors.cyan,
              child: Text('I am the text', style: TextStyle(fontSize: 22),)),
        ),
      ),
    );
  }
}
