  State<MyHomePage> createState() => _MyHomePageState();

class _MyHomePageState extends State<MyHomePage> {

  Widget build(BuildContext context) {
    return DefaultTabController(

import 'package:flutter/material.dart';
import 'package:shikar/screens/chats/calls_for_tabbar_file.dart';
import 'package:shikar/screens/chats/chats_for_tabbar_file.dart';
import 'package:shikar/screens/chats/status_for_tabbarfile.dart';

class Tabbarfile extends StatefulWidget {
  const Tabbarfile({super.key});

  @override
  State<Tabbarfile> createState() => _TabbarfileState();
}

class _TabbarfileState extends State<Tabbarfile> {
  @override
  Widget build(BuildContext context) {
    return DefaultTabController(
      length: 3, 
      child: Scaffold(
        appBar: AppBar(
          title: Text("Tab bar page", style: TextStyle(color: Colors.white),),
          backgroundColor: Colors.green,
          bottom: TabBar(
            labelColor: Colors.white,
            unselectedLabelColor: Colors.grey,
            indicatorColor: Colors.white,
            tabs: [
                Tab(text: "CHAT",icon: Icon(Icons.chat),),
                Tab(text: "STATUS",icon: Icon(Icons.message),),
                Tab(text: "CALL",icon: Icon(Icons.call),),

            ]),

          
        ),

        body: TabBarView(
          children: [
                    ChatScreen(),
                    StatusScreen(),
                    CallScreen()

          ]),
      )
      );
  }
}
