Here if you press the button , nothing will happen because when button will be pressed, setstate function will be called.

7. Date picker concept -

var datePicked = showDatePicker(
       context: context, 
	   initialDate: initialDate, ------->It will show current date to pick
       firstDate: firstDate, --------->we can not go back to this date
       lastDate: lastDate------------> we can not go beyond/future to  this date
	   )
	   
	   
	   
=============

var datePicked = showDatePicker() ---------->The type is not defined.

DateTime datePicked = showDatePicker()------>The type is DateTime. but to display the value it will take time. 
                                             so other program should not wait for this. so use await.


DateTime datePicked = await showDatePicker()


ElevatedButton(onPressed: () async {DateTime datePicked = await showDatePicker()};--->use async.

ElevatedButton(onPressed: () async {DateTime ? datePicked = await showDatePicker()}; --->if you select outside of datepicket diaolg box, it will be null. so use ?.

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flutter Demo',
      theme: ThemeData(

        colorScheme: ColorScheme.fromSeed(seedColor: Colors.deepPurple),
        useMaterial3: true,
      ),
      home: const MyHomePage(title: 'Flutter Demo Home Page'),
    );
  }
}

class MyHomePage extends StatefulWidget {
  const MyHomePage({super.key, required this.title});



  final String title;

  @override
  State<MyHomePage> createState() => _MyHomePageState();
}

class _MyHomePageState extends State {
  @override
  Widget build(BuildContext context) {

    return Scaffold(
      appBar: AppBar(

        title: Text('Hello'),
      ),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Text(
              'Select Date',
              style: TextStyle(fontSize: 15),
            ),
            ElevatedButton(onPressed: () async {
              DateTime? datePicked =  await showDatePicker(
                context: context,
                //initialDate: DateTime.now(),
                initialDate: DateTime.now(),
                firstDate: DateTime(2021),
                lastDate: DateTime.now(),
              );
              if(datePicked!=null){
                print('Date selected: ${datePicked.day}-${datePicked.month}-${datePicked.year}');
              }

            }, child: Text('Show'))
          ],
        ),
      ),

    );
  }
}

9. Time picker

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flutter Demo',
      theme: ThemeData(

        colorScheme: ColorScheme.fromSeed(seedColor: Colors.deepPurple),
        useMaterial3: true,
      ),
      home: const MyHomePage(title: 'Flutter Demo Home Page'),
    );
  }
}

class MyHomePage extends StatefulWidget {
  const MyHomePage({super.key, required this.title});



  final String title;

  @override
  State createState() => _MyHomePageState();
}

class _MyHomePageState extends State {
  @override
  Widget build(BuildContext context) {

    return Scaffold(
      appBar: AppBar(

        title: Text('Hello'),
      ),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Text(
              'Select Date',
              style: TextStyle(fontSize: 15),
            ),
            ElevatedButton(onPressed: () async {
              DateTime? datePicked =  await showDatePicker(
                context: context,
                //initialDate: DateTime.now(),
                initialDate: DateTime.now(),
                firstDate: DateTime(2021),
                lastDate: DateTime.now(),
              );
              if(datePicked!=null){
                print('Date selected: ${datePicked.day}-${datePicked.month}-${datePicked.year}');
              }

            }, child: Text('Show')),
            ElevatedButton(onPressed: () async {
              TimeOfDay? pickedTime = await showTimePicker(
                  context: context,
                  initialTime: TimeOfDay.now(),
                 initialEntryMode: TimePickerEntryMode.input
              );

              if(pickedTime!=null){
                print('Time selected: ${pickedTime.hour}-${pickedTime.minute}');
              }

            }, child: Text('Select Time'))
          ],
        ),
      ),);
  }
}
