Callback Function-

1. Example of an Elevated Button -

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {

    return MaterialApp(
        title: 'Flutter Demo',
        home: Scaffold(
          appBar: AppBar(
            backgroundColor: Colors.purple,
            title: Text('Hello'),
          ),
          body: ElevatedButton(
            child: Text('Click me'),
            onPressed: () {
              print('I am clicked!!!!!!!!!');
            },
          )
        ));
  }
}

2. Use callback function

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {

    callback1() {
      print('I am clicked!!!!!!!!!');
    }

    return MaterialApp(
        title: 'Flutter Demo',
        home: Scaffold(
          appBar: AppBar(
            backgroundColor: Colors.purple,
            title: Text('Hello'),
          ),
          body: ElevatedButton(
            child: Text('Click me'),
            onPressed: callback1,
          )
        ));
  }
}


create one function - callback1 and we can use it by calling from anywhere.
