Display a widgt once click the button

import 'package:flutter/material.dart';
import 'package:loading_animation_widget/loading_animation_widget.dart';

class Test123 extends StatefulWidget {
  const Test123({super.key});

  @override
  State<Test123> createState() => _Test123State();
}

class _Test123State extends State<Test123> {
  bool _showContainer = false;-------------->declare a variable
  @override
  Widget build(BuildContext context) {
    return Scaffold(
            body:  Center(
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  ElevatedButton(
                    onPressed: () {
                      // Update state to show container
                      setState(() {
                        _showContainer = true;
                      });
                    },
                    child: Text('Show Colored Container'),
                  ),
                  SizedBox(height: 20),
                  // Show container only when _showContainer is true
                  if (_showContainer)
                    Container(
                      width: 200,
                      height: 200,
                      color: Colors.blue, // Change the color as desired
                    ),
                ],
              ),
            ),
    );
  }
}

Display an alert box once click the button

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(
          title: Text('Colored Container Example'),
        ),
        body: Center(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              ElevatedButton(
                onPressed: () {
                  // Call a function to display the colored container
                  _showColoredContainer(context);
                },
                child: Text('Show Colored Container'),
              ),
            ],
          ),
        ),
      ),
    );
  }

  void _showColoredContainer(BuildContext context) {
    // Show a dialog with a colored container
    showDialog(
      context: context,
      builder: (BuildContext context) {
        return AlertDialog(
          title: Text('Colored Container'),
          content: Container(
            width: 200,
            height: 200,
            color: Colors.blue, // Change the color as desired
          ),
          actions: [
            TextButton(
              onPressed: () {
                Navigator.of(context).pop(); // Close the dialog
              },
              child: Text('Close'),
            ),
          ],
        );
      },
    );
  }
}
