Custom widget

1. Create the file to create custom widget-

flutapp1/base/widgets/app_double_text.dart
======================================
import 'package:flutapp1/base/res/styles/app_styles.dart';
import 'package:flutter/material.dart';

class AppDoubleText extends StatelessWidget {
  const AppDoubleText({super.key, required this.bigtext, required this.smalltext, });
  final String bigtext ;
  final String smalltext ;

  
  

  @override
  Widget build(BuildContext context) {
    return Row(
      mainAxisAlignment: MainAxisAlignment.spaceBetween,
      children: [
        //Text(bigtext, style: const TextStyle(fontSize: 20, fontWeight: FontWeight.w500),),
        Text(bigtext, style: AppStyles.headLineStyle2),
        
        InkWell(
          onTap: (){
            print("You are tapped");
          },
          //child: Text(smalltext, style: AppStyles.textStyle),
          child: Text(smalltext, style: AppStyles.textStyle.copyWith(color: AppStyles.primaryColor)),
		  //If we want to overwrite the feature , then we need to use copywith.
        )
        
      ],
    );
  }
}
